% Keyboard Navigation Implementation

This document presents the technical implementation of keyboard navigation features developed to enhance accessibility within the application's editor and app builder environments. The implementation enables comprehensive keyboard-based interaction with UI components, ensuring compliance with WCAG 2.1 Level AA accessibility standards.

\subsubsection{Component Accessibility}

\textbf{Making Components Focusable with Enter Key Support}

The implementation ensures that interactive components within the dynamic properties panel receive proper keyboard focus and respond to Enter key activation, mirroring mouse click functionality.

\textbf{Button Component}

The Button component integrates native HTML focus attributes and keyboard event handlers to enable tab navigation and Enter key activation.

\begin{lstlisting}[caption=Button Component Focusability Implementation, label=lst:button-focus]
// File: frontend/src/Editor/Components/Button.jsx

const handleKeyDown = (e) => {
  if (e.key === 'Enter' || e.key === ' ') {
    e.preventDefault();
    if (canClick) {
      fireEvent('onClick');
    }
  }
};

return (
  <button
    className={cx(`btn btn-${type}`)}
    style={computedStyles}
    onClick={() => canClick && fireEvent('onClick')}
    onKeyDown={handleKeyDown}
    disabled={disable}
    data-cy={dataCy}
    tabIndex={disable ? -1 : 0}
    role="button"
    aria-label={label}
  >
    {loading && <Loader style={{ '--loader-color': computedLoaderColor }} />}
    {!loading && label}
  </button>
);
\end{lstlisting}

The implementation utilizes \texttt{tabIndex=\{0\}} to include the button in the sequential keyboard navigation order, while \texttt{tabIndex=\{-1\}} removes disabled buttons from the tab sequence. The \texttt{onKeyDown} handler captures both Enter and Space key presses, preventing default browser behavior and invoking the onClick event handler.

\textbf{Dynamic Component Properties}

Dynamic components within the app builder's properties panel are made focusable through systematic DOM traversal and attribute injection.

\begin{lstlisting}[caption=Dynamic Component Focusability, label=lst:dynamic-focus]
// File: frontend/src/_components/KeyboardNavigation/KeyboardNavigation.jsx

const makeElementsFocusable = useCallback(() => {
  const fileListItems = document.querySelectorAll(`
    .list-group-item,
    [data-cy*="list-card"],
    [role="button"],
    .application-card
  `);

  fileListItems.forEach(item => {
    if (item.hasAttribute('tabindex') || item.hasAttribute('disabled')) 
      return;
    
    const style = window.getComputedStyle(item);
    const rect = item.getBoundingClientRect();

    if (style.display !== 'none' && 
        style.visibility !== 'hidden' &&
        rect.width > 0 && rect.height > 0) {
      item.setAttribute('tabindex', '0');
      item.classList.add('keyboard-navigable');
      
      // Add Enter key handler
      item.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          item.click();
        }
      });
    }
  });
}, []);
\end{lstlisting}

This approach ensures that dynamically rendered components receive keyboard focus capabilities without requiring individual component modifications, maintaining consistency across the application.

\subsubsection{Keyboard-Based Interactions}

\textbf{Drag and Drop Keyboard Support}

Keyboard-based component placement provides an alternative to mouse-based drag-and-drop operations, enabling users to add components to the canvas using arrow keys for positioning and Enter for confirmation.

\begin{lstlisting}[caption=Keyboard Component Placement Implementation, label=lst:keyboard-dragdrop]
// File: frontend/src/AppBuilder/RightSideBar/ComponentManagerTab/DragLayer.jsx

const handleKeyDown = (e) => {
  if (disabled) return;

  if (e.key === 'Enter' || e.key === ' ') {
    e.preventDefault();
    
    // Start keyboard placement mode
    const startKeyboardPlacement = useStore.getState().startKeyboardPlacement;
    startKeyboardPlacement(component);
    
    // Close sidebar if not pinned
    if (!isRightSidebarPinned) {
      toggleRightSidebar(false);
    }
  }
};

// File: frontend/src/AppBuilder/AppCanvas/KeyboardPlacementOverlay.jsx

const handleKeyDown = useCallback((e) => {
  if (!active) return;

  // Arrow keys - move the component preview
  if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(e.key)) {
    e.preventDefault();
    moveKeyboardPlacementPosition(e.key);
    return;
  }

  // Enter - place the component at current position
  if (e.key === 'Enter') {
    e.preventDefault();
    if (component) {
      const item = { componentType: component.component, component };
      handleDrop(item, 'canvas');
      cancelKeyboardPlacement();
    }
    return;
  }

  // Escape - cancel placement
  if (e.key === 'Escape') {
    e.preventDefault();
    cancelKeyboardPlacement();
    return;
  }
}, [active, component]);
\end{lstlisting}

The keyboard placement system renders a visual overlay showing the component's preview position, which updates in real-time as users press arrow keys. This provides immediate visual feedback for precise positioning.

\textbf{Component Resizing Shortcuts}

Components support keyboard-based resizing through modifier keys combined with arrow keys, enabling precise dimensional adjustments without mouse interaction.

\begin{lstlisting}[caption=Keyboard Resizing Implementation, label=lst:keyboard-resize]
// File: frontend/src/AppBuilder/AppCanvas/WidgetWrapper.jsx

const [resizeModifier, setResizeModifier] = useState(null);

const handleKeyDown = (e) => {
  if (!isWidgetActive || readOnly) return;

  // = key for expanding
  if (e.key === '=' || e.code === 'Equal') {
    e.preventDefault();
    setResizeModifier('expand');
    setIsKeyboardResizing(true);
    return;
  } 
  // - key for shrinking
  else if (e.key === '-' || e.code === 'Minus') {
    e.preventDefault();
    setResizeModifier('shrink');
    setIsKeyboardResizing(true);
    return;
  }

  // Arrow keys with active modifier
  if (isArrowKey && resizeModifier) {
    e.preventDefault();
    e.stopPropagation();
    
    const { resizeComponentWithKeyboard } = useStore.getState();
    resizeComponentWithKeyboard(e.key, resizeModifier === 'expand');
    
    // Screen reader feedback
    const action = resizeModifier === 'expand' ? 'Expanding' : 'Shrinking';
    const direction = e.key.replace('Arrow', '').toLowerCase();
    speak(`${action} ${componentName} ${direction}`);
  }
};
\end{lstlisting}

The resizing mechanism prevents movement operations when a resize modifier is active, ensuring that arrow key presses modify dimensions rather than position.

\textbf{Canvas Scroll Navigation}

Ctrl+Arrow key combinations control canvas viewport scrolling independent of component movement or focus changes.

\begin{lstlisting}[caption=Canvas Scroll Keyboard Control, label=lst:keyboard-scroll]
// File: frontend/src/_components/KeyboardNavigation/KeyboardNavigation.jsx

useHotkeys('control+up, control+down, control+left, control+right', 
  (e) => {
    e.preventDefault();
    
    const canvas = document.querySelector('.real-canvas');
    if (!canvas) return;
    
    const scrollAmount = 50; // pixels
    
    switch(e.key) {
      case 'ArrowUp':
        canvas.scrollTop -= scrollAmount;
        break;
      case 'ArrowDown':
        canvas.scrollTop += scrollAmount;
        break;
      case 'ArrowLeft':
        canvas.scrollLeft -= scrollAmount;
        break;
      case 'ArrowRight':
        canvas.scrollLeft += scrollAmount;
        break;
    }
    
    speak(`Scrolled canvas ${e.key.replace('Arrow', '').toLowerCase()}`);
  }, 
  { enableOnTags: ['INPUT', 'TEXTAREA', 'SELECT'] }
);
\end{lstlisting}

The scroll navigation remains functional even when input fields have focus, utilizing the \texttt{enableOnTags} option to override default behavior for these specific shortcuts.

\subsubsection{Navigation Patterns}

\textbf{Arrow Key Navigation in Tab Options}

Tab-based interfaces within settings menus implement arrow key navigation for rapid movement between options, following ARIA authoring practices for tab panels.

\begin{lstlisting}[caption=Tab Options Arrow Key Navigation, label=lst:arrow-tabs]
// File: frontend/src/TooljetDatabase/Forms/RowForm.jsx

const handleTabKeyDown = (e) => {
  if (e.key === 'ArrowLeft' || e.key === 'ArrowRight') {
    e.preventDefault();
    e.stopPropagation();
    
    const tablist = e.currentTarget.parentElement;
    const allTabs = Array.from(
      tablist.querySelectorAll('[role="button"]')
    );
    const currentIdx = allTabs.indexOf(e.currentTarget);
    
    const nextIdx = e.key === 'ArrowRight'
      ? (currentIdx + 1) % allTabs.length
      : (currentIdx - 1 + allTabs.length) % allTabs.length;
    
    allTabs[nextIdx]?.focus();
    
    // Announce new tab
    speak(`${allTabs[nextIdx].textContent} tab`);
  } else if (e.key === 'Enter' || e.key === ' ') {
    e.preventDefault();
    speak(`${e.currentTarget.textContent} tab selected`);
    handleTabClick(e.currentTarget.dataset.tabIndex);
  }
};

return (
  <div role="tablist" aria-label="Options">
    {tabs.map((tab, index) => (
      <div
        key={index}
        role="button"
        tabIndex={activeTab === index ? 0 : -1}
        aria-selected={activeTab === index}
        onKeyDown={handleTabKeyDown}
        onClick={() => handleTabClick(index)}
      >
        {tab.label}
      </div>
    ))}
  </div>
);
\end{lstlisting}

The tab navigation implements roving tabindex pattern, where only the active tab is in the tab sequence (\texttt{tabIndex=\{0\}}), while inactive tabs are removed (\texttt{tabIndex=\{-1\}}). Arrow keys move focus and update the roving tabindex.

\textbf{Dynamic Component Navigation}

The global keyboard navigation system automatically detects and integrates dynamically rendered components into the navigation sequence through continuous DOM observation.

\begin{lstlisting}[caption=Dynamic Component Navigation Integration, label=lst:dynamic-navigation]
// File: frontend/src/_components/KeyboardNavigation/KeyboardNavigation.jsx

const getFocusableElements = useCallback(() => {
  const elements = [];
  
  // Sidebar navigation
  const sidebarSelectors = [
    '.tj-leftsidebar-icon-items[data-cy="icon-dashboard"]',
    '.tj-leftsidebar-icon-items[data-cy="icon-workflows"]',
    '.tj-leftsidebar-icon-items[data-cy="icon-database"]',
  ];
  
  sidebarSelectors.forEach(selector => {
    const el = document.querySelector(selector);
    if (el && isElementVisible(el)) {
      elements.push(el);
    }
  });
  
  // Dynamic component cards
  const cards = document.querySelectorAll('.app-card, .homepage-app-card');
  cards.forEach(card => {
    if (isElementVisible(card)) {
      elements.push(card);
    }
  });
  
  // Make new elements focusable
  elements.forEach(el => {
    if (!el.hasAttribute('tabindex')) {
      el.setAttribute('tabindex', '0');
      el.classList.add('keyboard-navigable');
    }
  });
  
  return elements;
}, [isElementVisible]);

// Update navigation elements when DOM changes
useEffect(() => {
  const observer = new MutationObserver(() => {
    getFocusableElements();
  });
  
  observer.observe(document.body, {
    childList: true,
    subtree: true
  });
  
  return () => observer.disconnect();
}, [getFocusableElements]);
\end{lstlisting}

The MutationObserver API monitors DOM changes, triggering updates to the focusable element list when components are added or removed, ensuring navigation sequences remain current.

\subsubsection{Focus Management}

\textbf{Focus Traps with ESC Key}

Modal dialogs and dropdown menus implement focus trapping to constrain keyboard navigation within the active context, preventing focus from escaping to background elements.

\begin{lstlisting}[caption=Menu Focus Trap Implementation, label=lst:focus-trap]
// File: frontend/src/TooljetDatabase/TableListItem/ActionsPopover/index.jsx

useEffect(() => {
  if (!show) return;

  // Focus first menu item
  setTimeout(() => {
    const firstOption = document.querySelector(
      '.table-list-items [tabindex="0"]'
    );
    if (firstOption) {
      firstOption.focus();
    }
  }, 100);

  // Focus trap keyboard handler
  const handleKeyDown = (e) => {
    if (!document.activeElement?.closest('.table-list-items')) 
      return;

    const getFocusableElements = () => {
      const popup = document.querySelector('.table-list-items');
      if (!popup) return [];
      
      const selectors = [
        '[tabindex="0"]',
        'button:not([disabled])',
        'a[href]:not([disabled])'
      ].join(', ');
      
      return Array.from(popup.querySelectorAll(selectors))
        .filter(el => el.offsetParent !== null);
    };

    if (e.key === 'Tab') {
      e.preventDefault();
      e.stopPropagation();
      
      const items = getFocusableElements();
      const currentIndex = items.indexOf(document.activeElement);
      
      // Shift+Tab: previous item (wrap around)
      // Tab: next item (wrap around)
      const nextIndex = e.shiftKey 
        ? (currentIndex - 1 + items.length) % items.length
        : (currentIndex + 1) % items.length;
      
      items[nextIndex]?.focus();
      
    } else if (e.key === 'Escape') {
      e.preventDefault();
      speak('Closing menu');
      setTimeout(() => {
        setShow(false);
        // Return focus to trigger button
        triggerButtonRef.current?.focus();
      }, 100);
    }
  };

  // Use capture phase for priority handling
  document.addEventListener('keydown', handleKeyDown, true);
  return () => document.removeEventListener('keydown', handleKeyDown, true);
}, [show]);
\end{lstlisting}

Focus traps prevent Tab from moving focus outside the menu, cycling through menu items instead. Escape closes the menu and returns focus to the triggering element, maintaining user context.
